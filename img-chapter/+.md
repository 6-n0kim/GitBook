# 얼굴 + 번호판 탐지 후 바운딩 박스 그리기

### **2. 객체 탐지 & 바운딩 박스 그리기**

1. 이미지에서 MTCNN(혹은 YOLO) 같은 모델을 사용해 얼굴(또는 원하는 객체)을 찾아, 원본 위에 빨간색 테두리 박스를 그립니다.
2. 이 과정은 이후 블러/마스킹 처리할 **대상 영역을 식별**하는 단계입니다.

```python
import cv2
import numpy as np
from facenet_pytorch import MTCNN
from ultralytics import YOLO
import torch

# 얼굴 탐지 함수 (MTCNN 사용)
mtcnn = MTCNN(keep_all=True, # 다중 탐지 허용
              device='cuda' if torch.cuda.is_available() else 'cpu')

def detect_faces(img_pil):
    boxes, _ = mtcnn.detect(img_pil) # (x1,y1,x2,y2)
    return [tuple(map(int, b)) for b in boxes] if boxes is not None else [] # 없으면 빈 리스트, 좌표는 int로 변환

# 번호판 탐지 함수 (YOLOv8)
plate_model = YOLO('license_plate_yolov8.pt')
def detect_plate_boxes(img_pil):
    results = plate_model(img_pil)
    preds = results[0].boxes.data  # yolo결과에서 박스들만 추출하고 실제 숫자 데이터를 가져옴: x1,y1,x2,y2,conf,class
    if preds is None or len(preds) == 0:
        return []
    boxes = preds[:, :4].cpu().numpy().astype(int) # 좌표만 추출 → numpy → int
    return [tuple(b) for b in boxes] # 리스트[ (x1,y1,x2,y2), ... ] 형태로 반환

# 탐지 실행
face_boxes = detect_faces(image) # 얼굴 바운딩 박스 목록
plate_boxes = detect_plate_boxes(image) # 번호판 바운딩 박스 목록

# 이미지에 박스 그리기
img_vis = image.copy()
draw = ImageDraw.Draw(img_vis)
font = ImageFont.load_default()

# 얼굴 박스 그리기
for (x0, y0, x1, y1) in face_boxes:
    draw.rectangle([x0, y0, x1, y1], outline='red', width=2)
    draw.text((x0, y0 - 10), "FACE", fill='red', font=font)

# 번호판 박스 그리기
for (x0, y0, x1, y1) in plate_boxes:
    draw.rectangle([x0, y0, x1, y1], outline='blue', width=2)
    draw.text((x0, y0 - 10), "PLATE", fill='blue', font=font)

# 결과 출력

plt.show()
```
