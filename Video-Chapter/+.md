# 얼굴 + 번호판 탐지 후 바운딩 박스 그리기

### **3. 객체 탐지 & 바운딩 박스 그리기**

*
  1. 이미지에서 MTCNN(혹은 YOLO) 같은 모델을 사용해 얼굴(또는 원하는 객체)을 찾아, 원본 위에  박스를 그립니다.
  2. 이 과정은 이후 블러/마스킹 처리할 **대상 영역을 식별**하는 단계입니다.

```python
import cv2
import numpy as np
from facenet_pytorch import MTCNN
from ultralytics import YOLO
import torch
from tqdm import tqdm

# 모델 초기화
mtcnn = MTCNN(keep_all=True, device='cuda' if torch.cuda.is_available() else 'cpu')
plate_model = YOLO('license_plate_yolov26.pt')

# 비디오 열기
cap = cv2.VideoCapture(video_path) # 비디오 파일 핸들(열쇠) 생성
original_frames = []               # 원본 프레임(BGR) 저장 리스트
frames_with_boxes = []             # 박스가 그려진 프레임(RGB np.array) 저장 리스트
face_box_list = []                 # 프레임별 얼굴 박스 좌표 목록
plate_box_list = []                # 프레임별 번호판 박스 좌표 목록

while True:
  ret, frame = cap.read()
  if not ret:
    break

  # opencv -> PIL변환
  image_pil = Image.fromarray(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)) # BGR->RGB 후 PIL Image로
  draw = ImageDraw.Draw(image_pil)
  font = ImageFont.load_default()

  # 1) 얼굴 탐지 함수 (MTCNN 사용)
  face_boxes, _ = mtcnn.detect(image_pil) # PIL 이미지 입력 → 얼굴 박스(np.array)
  face_boxes = [] if face_boxes is None else face_boxes.astype(int).tolist() # None 처리 및 int 좌표 리스트로 변환

  # 2) 번호판 탐지 함수 (YOLOv26)
  results = plate_model(np.array(image_pil)) # PIL 이미지를 RGB numpy로 변환해 YOLO 추론
  plate_preds = results[0].boxes.data  # tensor: x1,y1,x2,y2,conf,class
  plate_boxes = [] if plate_preds is None or len(plate_preds) == 0 else plate_preds[:, :4].cpu().numpy().astype(int).tolist()

  # 얼굴 박스 그리기
  for (x0, y0, x1, y1) in face_boxes:
      draw.rectangle([x0, y0, x1, y1], outline='red', width=2)
      draw.text((x0, y0 - 10), "FACE", fill='white', font=font)

  # 번호판 박스 그리기
  for (x0, y0, x1, y1) in plate_boxes:
      draw.rectangle([x0, y0, x1, y1], outline='blue', width=2)
      draw.text((x0, y0 - 10), "PLATE", fill='white', font=font)

  # PIL -> RGB numpy 이미지로 저장
  processed_frame = np.array(image_pil)
  frames_with_boxes.append(processed_frame)
  face_box_list.append(face_boxes)
  plate_box_list.append(plate_boxes)
  original_frames.append(frame.copy())

cap.release() # 비디오 핸들 해제
print(f"총 {len(frames_with_boxes)}개 프레임 처리 완료.")

# 슬라이더 시각화
def show_frame(idx):
  plt.figure(figsize=(8, 6))
  plt.imshow(frames_with_boxes[idx])
  plt.axis("off")
  plt.show()

# 슬라이더 UI 만들기
slider = widgets.IntSlider(value=0, min=0, max=len(frames_with_boxes)-1, step=1, description='Frame:')
widgets.interact(show_frame, idx=slider)
```
